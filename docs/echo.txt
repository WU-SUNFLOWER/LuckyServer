void Channel::HandleEvent()
{
    if (ready_ & (EPOLLIN | EPOLLPRI))
    {
        read_callback_();
    }
    if (ready_ & EPOLLOUT) //于channel.cpp文件的第34行。
    {
        write_callback_();
    }
}

void EventLoop::Loop()
{
    while (!quit_)
    {
        std::vector<Channel *> channels = epoll_->Wait();
        for (Channel *channel : channels)
        {
            channel->HandleEvent();
        }
    }
}
